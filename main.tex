% English main font (NTU: Times New Roman).
% - Strict mode (FINAL): stop if Times New Roman is not available.
% - Draft mode: fall back to TeX Gyre Termes for preview (NOT submission-safe).
\makeatletter
\newcommand{\NTU@SetMainFont}{%
  \ifx\MainFontFile\@empty
    \IfFontExistsTF{\MainFontName}{%
      \setmainfont{\MainFontName}%
    }{%
      \ifStrictNTUFonts
        \errmessage{Times New Roman not found.\MessageBreak
        FINAL mode requires Times New Roman.\MessageBreak
        Overleaf fix: upload the font file(s) and set \string\def\string\MainFontFile{fonts/TimesNewRoman.ttf} in config.tex,\MessageBreak
        OR set \string\StrictNTUFontsfalse for DRAFT preview (not submission-safe).}%
      \else
        \typeout{NTU WARNING: Times New Roman not found; using TeX Gyre Termes for DRAFT preview.}%
        \setmainfont{TeX Gyre Termes}%
      \fi
    }%
  \else
    \IfFileExists{\MainFontFile}{%
      \setmainfont{\MainFontFile}%
    }{%
      \ifStrictNTUFonts
        \errmessage{MainFontFile not found: \MainFontFile\MessageBreak
        Upload the font file to the project (e.g., fonts/TimesNewRoman.ttf), or clear MainFontFile to use MainFontName.}%
      \else
        \typeout{NTU WARNING: MainFontFile not found; using TeX Gyre Termes for DRAFT preview.}%
        \setmainfont{TeX Gyre Termes}%
      \fi
    }%
  \fi
}
\makeatother

      {\fontsize{18pt}{27pt}\selectfont \CoverDegreeCH\par}

      {\fontsize{14pt}{21pt}\selectfont \CoverInstituteEN\par}
      {\fontsize{14pt}{21pt}\selectfont \CoverCollegeEN\par}
      {\fontsize{16pt}{24pt}\selectfont \CoverUniversityEN\par}
      {\fontsize{16pt}{24pt}\selectfont \CoverDegreeEN\par}
      \vspace{2cm}
      {\fontsize{18pt}{27pt}\selectfont \ThesisTitleCH \par}
      {\fontsize{18pt}{27pt}\selectfont \ThesisTitleEN \par}
      \vfill
      {\fontsize{18pt}{27pt}\selectfont \AuthorCH \par}
      {\fontsize{18pt}{27pt}\selectfont \AuthorEN \par}
      \vspace{1.5cm}
      {\fontsize{18pt}{27pt}\selectfont 指導教授：\AdvisorCH \par}
      {\fontsize{18pt}{27pt}\selectfont Advisor: \AdvisorEN \par}
